' ..----------------------------------------------------------------------------
' --- TEST_Debughelper ---
Sub TEST_Debughelper()
    Dim ENGINE_PATH As String
    Dim sessionId As String
    Dim result As String

    ' --- 実行環境の初期化 ---
    ENGINE_PATH = ThisWorkbook.Path & PS_ENGINE_LIB
    sessionId = "Debug " & Format(Now, "yyyyMMdd_HHmmss")
    Set rpaEngine = New Ps_Engine
    
    Debug.Print "=== 処理開始: " & sessionId & " ==="
    Dim useCdpPort As Integer ' 通信モード (0: 標準, 9222等: CDP)
    useCdpPort = 0
    ' --- エンジンの起動および接続確認 ---
    If Not rpaEngine.StartEngine(sessionId, ENGINE_PATH, useCdpPort, True) Then
        MsgBox "エンジンの起動に失敗しました。", vbCritical
        Exit Sub
    End If
    
    On Error GoTo ErrorHandler
' ..----------------------------------------------------------------------------
' --- ▼ P01（画面）
    ' テストサイト：www.yahoo.co.jp
    rpaEngine.RunAction "Invoke-WebNavigation", CreateParams("Url", "https://●●●.yah●●.co.jp/")
  
    ' エラーを無視して強行突破する（ページロード待機での広告等のタイムアウト）
    On Error Resume Next
    rpaEngine.RunAction "Wait-WebPageLoad", CreateParams("TimeoutSec", 15)
    ' エラーをリセット
    Err.Clear
     ' エラー監視を元に戻す
    On Error GoTo ErrorHandler
' ..----------------------------------------------------------------------------
    ' --- モードレス（非同期）でフォームを表示 ---
    UserForm1.Show vbModeless
    loopCount = 1

    ' 非同期監視ループ
    Do
        If g_DebugAction = 1 Then ' [エクスポート実行] ボタン
            Debug.Print "--- TEST_Debughelper / loopCount: " & Format(loopCount, "00") & " Note: " & g_DebugNote
            g_DebugAction = 0 ' フラグをリセット
            Call Debug_ExportLoop(loopCount)
            UserForm1.lblStatus.Caption = "T" & Format(loopCount, "00") & " 出力完了。"
            UserForm1.TextNote = "ｘｘｘ"
            loopCount = loopCount + 1
        End If
        
        If g_DebugAction = 9 Then ' [デバッグ終了] ボタン
            Exit Do
        End If
        
        DoEvents
        ' 大きくすると、文字入力がストレス
        Sleep 50
    Loop
    
' ..----------------------------------------------------------------------------
    Debug.Print "--- キャッシュクリア実行 ---"
    rpaEngine.RunAction "Clear-WebCache", CreateParams("Mode", "CacheOnly")
    DoEvents
    rpaEngine.RunAction "Clear-WebCache", CreateParams("Mode", "All")
    
' --- メモリ解放 ---
CleanUp:
    If Not rpaEngine Is Nothing Then
        rpaEngine.CloseEngine
    End If
    Set rpaEngine = Nothing
    Exit Sub
    
' --- 異常系のハンドリング（エラー捕捉） ---
ErrorHandler:
    If Err.Source = "PS_Engine" Then
        ' 例外情報を格納する構造体（エラー文字列をパース）
        Dim rpaErr As RpaExceptionInfo
        rpaErr = ParseRpaError(Err.Description)

        Select Case rpaErr.ErrorType
            Case "未発見"
                Debug.Print "【スキップ】要素が見つかりません: " & rpaErr.Details
                Resume Next
            Case "Timeout"
                Debug.Print "【待機超過】画面の応答がありません (" & rpaErr.FunctionName & ")"
            Case "JSエラー", "ネイティブエラー", "内部エラー"
                MsgBox "ブラウザ制御内で致命的なエラーが発生しました。" & vbCrLf & _
                       "関数: " & rpaErr.FunctionName & vbCrLf & _
                       "内容: " & rpaErr.Message, vbCritical
            Case Else
                Debug.Print "【その他エラー】" & rpaErr.RawText
        End Select
    Else
        MsgBox "VBAマクロエラー (" & Err.Number & "): " & Err.Description, vbCritical
    End If
    
    On Error Resume Next
    MsgBox "*** エラーが発生しました:" & vbCrLf & Err.Description, vbCritical
    Resume CleanUp
End Sub

' --- 最前面/解除を切り替える共通関数 ---
Private Sub SetTopMost(ByVal isTop As Boolean)
    Dim hwnd As LongPtr
    hwnd = FindWindow("ThunderDFrame", Me.Caption)
    
    If hwnd <> 0 Then
        If isTop Then
            ' 最前面にピン留め
            Call SetWindowPos(hwnd, HWND_TOPMOST, 0, 0, 0, 0, SWP_NOMOVE Or SWP_NOSIZE)
        Else
            ' ピン留め解除（裏に回れるようになる）
            Call SetWindowPos(hwnd, HWND_NOTOPMOST, 0, 0, 0, 0, SWP_NOMOVE Or SWP_NOSIZE)
        End If
    End If
End Sub

' --- タブ一覧の取得とリストボックスへの表示 ---（テスト用）
Private Sub btnGetTabs_Click()
    Dim resJson As String
    Dim jsonParsed As Object
    Dim item As Object
    
    Me.lblStatus.Caption = "タブ一覧を取得中..."
    DoEvents

    resJson = rpaEngine.RunAction("List-Tabs", CreateParams()) ' JSON文字列を受け取る
    lstTabs.Clear ' リストボックスを初期化 (2列設定: 0列目=Title, 1列目=TabId)
    lstTabs.ColumnCount = 2
    lstTabs.ColumnWidths = "150;0" ' 2列目(TabId)は非表示
    
    Set jsonParsed = JsonConverter.ParseJson(resJson) ' Collection([ ])で戻る
    If Not jsonParsed Is Nothing Then
        For Each item In jsonParsed
            lstTabs.AddItem item("Title")
            lstTabs.List(lstTabs.ListCount - 1, 1) = item("TabId")
        Next item
        Me.lblStatus.Caption = "タブ取得完了: " & lstTabs.ListCount & " 件"
    End If
End Sub

' --- 選択したタブへの切り替えとマスク待機 ---（テスト用）
Private Sub btnSwitchTab_Click()
    Dim targetTabId As String
    Dim targetTitle As String
    Dim maskXPath As String
    
    If lstTabs.ListIndex = -1 Then
        MsgBox "リストから切り替えたいタブを選択してください。", vbExclamation, "未選択"
        Exit Sub
    End If
    
    ' 選択された行のデータを取得
    targetTitle = lstTabs.List(lstTabs.ListIndex, 0)
    targetTabId = lstTabs.List(lstTabs.ListIndex, 1)
    Me.lblStatus.Caption = "タブ切替中..."
    DoEvents
    ' TabId を使ってフォーカスを切り替える
    rpaEngine.RunAction "Switch-Tab", CreateParams("TabId", targetTabId)
    Me.lblStatus.Caption = "マスク解除待機中..."
    DoEvents
    ' ロック（マスク）解除を待機
    rpaEngine.RunAction "Wait-WebScreenUnlock", CreateParams()
    ' Wait-WebScreenUnlockは汎用待機のため、補足的にSleepする
    Sleep 2000
    
    Me.lblStatus.Caption = "切替完了: " & targetTitle
    MsgBox "タブの切り替えとマスク待機が完了しました。" & vbCrLf & "次の操作（エクスポート等）が可能です。", vbInformation
    lstTabs.Clear
End Sub
